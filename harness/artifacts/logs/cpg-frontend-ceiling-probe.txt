import java.io.ByteArrayOutputStream;

/**
 * Does a java.io.ByteArrayOutputStream's growth ceiling move with the heap?
 *
 * flatgraph 0.1.32 serialises its string pool through one ByteArrayOutputStream.
 * This probe drives that class's own growth path past the soft maximum array
 * length and prints what it throws, so the failure the frontend hit can be
 * reproduced without the frontend and observed at two different heaps.
 */
public final class CeilingProbe {
    private static final int SOFT_MAX = Integer.MAX_VALUE - 8;   // 2,147,483,639

    public static void main(String[] args) {
        System.out.println("java.version      = " + System.getProperty("java.version"));
        System.out.println("java.vm.name      = " + System.getProperty("java.vm.name"));
        System.out.println("maxMemory bytes   = " + Runtime.getRuntime().maxMemory());
        System.out.println("SOFT_MAX          = " + SOFT_MAX);
        ByteArrayOutputStream out = new ByteArrayOutputStream(1 << 20);
        byte[] chunk = new byte[1 << 24];                        // 16 MiB
        long written = 0;
        try {
            while (written + chunk.length <= SOFT_MAX) {
                out.write(chunk, 0, chunk.length);
                written += chunk.length;
            }
            int remainder = (int) (SOFT_MAX - written);
            if (remainder > 0) { out.write(chunk, 0, remainder); written += remainder; }
            System.out.println("buffered bytes    = " + written + " (== SOFT_MAX: " + (written == SOFT_MAX) + ")");
            System.out.println("now writing 77 more bytes, exactly as the frontend's next string would");
            out.write(chunk, 0, 77);
            System.out.println("UNEXPECTED: the write succeeded, size = " + out.size());
        } catch (Throwable t) {
            System.out.println("THREW            = " + t.getClass().getName());
            System.out.println("MESSAGE          = " + t.getMessage());
            System.out.println("buffered bytes   = " + written);
        }
    }
}
